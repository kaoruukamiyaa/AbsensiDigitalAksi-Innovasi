/** * KODE APPS SCRIPT — DAFTAR HADIR DIGITAL AKSI (v10) * Menangani 4 hal: * 1. Simpan/baca PENGATURAN kegiatan per kegiatan (bisa banyak per tanggal) -> tab "Pengaturan" * 2. Simpan/baca/hapus/pindah DAFTAR PESERTA (roster + TTD) -> tab "Peserta" * 3. Catat riwayat KEHADIRAN (audit trail, append-only) -> tab "Kehadiran" (format lama, tetap jalan) * 4. Hapus SATU KEGIATAN (beserta pengaturan & seluruh pesertanya) -> action "deleteEvent" * * PERBAIKAN PENTING DI v10: * - "deleteEvent" SEBELUMNYA memakai rowEventId_() yang punya fallback * "kalau IDKegiatan kosong, anggap ID = Tanggal". Ini menyebabkan BUG * SERIUS: menghapus 1 kegiatan yang ID-nya kebetulan berupa tanggal polos * (mis. kegiatan default sebelum pernah diklik "+Kegiatan/Sesi Baru") ikut * menghapus SEMUA baris lama (tanpa IDKegiatan) di tanggal yang sama. * - Sekarang "deleteEvent" HANYA mencocokkan isi kolom IDKegiatan secara * PERSIS (exact match), tidak lagi pakai fallback tanggal. Baris lama tanpa * IDKegiatan (kolom benar-benar kosong) TIDAK akan pernah ikut terhapus * otomatis lewat menu ini — kalau perlu dihapus, hapus manual di sheet. * * PERUBAHAN DI v9: * - Tambah action "deleteEvent" (POST): menghapus baris pengaturan kegiatan * di tab "Pengaturan" DAN seluruh baris peserta terkait di tab "Peserta" * yang punya IDKegiatan (atau fallback Tanggal) yang sama. * * PERUBAHAN DI v8: * - Sebelumnya semua data (Pengaturan & Peserta) hanya dibedakan berdasarkan * kolom "Tanggal", sehingga kalau ada 2 kegiatan di TANGGAL YANG SAMA, * datanya saling menimpa satu sama lain. * - Sekarang ditambahkan kolom baru "IDKegiatan" (ID unik per kegiatan/sesi, * dibuat otomatis oleh halaman absensi) yang menjadi kunci utama untuk * menyimpan & mengambil data. Kolom "Tanggal" tetap berisi tanggal asli * kegiatan tsb (untuk ditampilkan di form & PDF). * - Data lama yang belum punya IDKegiatan tetap otomatis kebaca (fallback ke * kolom Tanggal sebagai ID), jadi TIDAK PERLU migrasi data manual. * - "getData" & payload lain sekarang memakai parameter/field "eventId" * (bukan lagi "date" sebagai kunci pencarian). Endpoint tetap menerima * "date" sebagai fallback untuk kompatibilitas versi lama. * * PERUBAHAN DI v7: * - Tambah action "listEvents" (GET): daftar semua kegiatan tersimpan. * PERUBAHAN DI v6: * - Tambah action "moveParticipantDate". * PERUBAHAN DI v5: * - Format link gambar TTD pakai "lh3.googleusercontent.com". * PERUBAHAN DI v3: * - Tab "Pengaturan" punya kolom "JamSelesai". * * CARA PASANG / UPDATE: * 1. Klik Deploy > Manage deployments > klik ikon pensil (Edit) pada deployment yang aktif. * 5. Di "Version", pilih "New version", lalu klik Deploy. * (Jangan buat deployment baru — edit yang sudah ada, supaya URL webhook TIDAK berubah) * 6. Kalau diminta otorisasi ulang, klik "Authorize access" dan ikuti langkahnya. * 7. Tidak perlu menjalankan fungsi migrasi apa pun untuk update ini — data lama * otomatis tetap terbaca lewat fallback Tanggal. * 8. Setelah deploy, hard refresh halaman absensinya (Ctrl+Shift+R) dan pastikan * file index.html yang dipakai juga sudah versi terbaru (yang mendukung * "eventId" & tombol hapus kegiatan), supaya frontend & backend cocok. */ var SIGNATURE_FOLDER_NAME = 'TTD Absensi AKSI'; function doGet(e) { var action = e.parameter.action; var eventId = e.parameter.eventId || e.parameter.date; // "date" = fallback versi lama var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === 'getData' && eventId) { return jsonResponse_({ settings: getSettingsForEvent_(ss, eventId), participants: getParticipantsForEvent_(ss, eventId) }); } if (action === 'listEvents') { return jsonResponse_({ events: listEvents_(ss) }); } return jsonResponse_({ error: 'Aksi tidak dikenali' }); } function doPost(e) { var body = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); if (body.action === 'saveSettings') { saveSettings_(ss, body); } else if (body.action === 'saveParticipant') { saveParticipant_(ss, body); } else if (body.action === 'deleteParticipant') { deleteParticipant_(ss, body); } else if (body.action === 'moveParticipantDate') { moveParticipantDate_(ss, body); } else if (body.action === 'deleteEvent') { deleteEvent_(ss, body); } else { saveAttendance_(ss, body); } return jsonResponse_({ success: true }); } // ================== HELPER ID KEGIATAN ================== // Ambil ID kegiatan dari satu baris: pakai kolom IDKegiatan kalau ada isinya, // kalau kosong (data lama) fallback pakai kolom Tanggal (diformat) sebagai ID — // ini sama persis dengan cara kerja versi lama sebelum ada IDKegiatan. function rowEventId_(row, tanggalColIdx, idKegiatanColIdx) { var id = row[idKegiatanColIdx]; if (id) return String(id); return formatDate_(row[tanggalColIdx]); } // ================== PENGATURAN KEGIATAN (tab "Pengaturan") ================== // Kolom: [Tanggal, Judul, Jam, Mode, Lokasi, IzinManualEntry, JamSelesai, IDKegiatan] var PENGATURAN_COL = { TANGGAL: 0, JUDUL: 1, JAM: 2, MODE: 3, LOKASI: 4, IZIN_MANUAL: 5, JAM_SELESAI: 6, ID_KEGIATAN: 7 }; function listEvents_(ss) { var sheet = ss.getSheetByName('Pengaturan'); if (!sheet) return []; var data = sheet.getDataRange().getValues(); var events = []; for (var i = 1; i < data.length; i++) { if (data[i][PENGATURAN_COL.TANGGAL]) { events.push({ eventId: rowEventId_(data[i], PENGATURAN_COL.TANGGAL, PENGATURAN_COL.ID_KEGIATAN), date: formatDate_(data[i][PENGATURAN_COL.TANGGAL]), title: data[i][PENGATURAN_COL.JUDUL] || '(Tanpa Judul)' }); } } events.sort(function(a, b) { return a.date < b.date ? 1 : (a.date > b.date ? -1 : 0); }); return events; } function getSettingsForEvent_(ss, eventId) { var sheet = ss.getSheetByName('Pengaturan'); if (!sheet) return null; var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (rowEventId_(data[i], PENGATURAN_COL.TANGGAL, PENGATURAN_COL.ID_KEGIATAN) === eventId) { return { eventId: eventId, date: formatDate_(data[i][PENGATURAN_COL.TANGGAL]), title: data[i][PENGATURAN_COL.JUDUL], time: data[i][PENGATURAN_COL.JAM], mode: data[i][PENGATURAN_COL.MODE], location: data[i][PENGATURAN_COL.LOKASI], allowManualEntry: data[i][PENGATURAN_COL.IZIN_MANUAL] === true || data[i][PENGATURAN_COL.IZIN_MANUAL] === 'TRUE', endTime: data[i][PENGATURAN_COL.JAM_SELESAI] || '' }; } } return null; } function saveSettings_(ss, body) { var sheet = ss.getSheetByName('Pengaturan'); if (!sheet) { sheet = ss.insertSheet('Pengaturan'); sheet.appendRow(['Tanggal', 'Judul', 'Jam', 'Mode', 'Lokasi', 'IzinManualEntry', 'JamSelesai', 'IDKegiatan']); } var eventId = body.eventId || body.date; var data = sheet.getDataRange().getValues(); var rowIndex = -1; for (var i = 1; i < data.length; i++) { if (rowEventId_(data[i], PENGATURAN_COL.TANGGAL, PENGATURAN_COL.ID_KEGIATAN) === eventId) { rowIndex = i + 1; break; } } var row = [body.date, body.title, body.time, body.mode, body.location, body.allowManualEntry, body.endTime || '', eventId]; if (rowIndex > 0) { sheet.getRange(rowIndex, 1, 1, row.length).setValues([row]); } else { sheet.appendRow(row); } } // Hapus SATU kegiatan: baris pengaturannya di tab "Pengaturan" DAN seluruh // baris peserta terkait di tab "Peserta". // // PENTING (perbaikan v10): dulu fungsi ini memakai rowEventId_() yang punya // fallback "kalau IDKegiatan kosong, anggap ID = Tanggal". Itu menyebabkan // baris LAMA (tanpa IDKegiatan) yang tanggalnya sama ikut IKUT TERHAPUS saat // admin menghapus satu kegiatan yang ID-nya juga hanya berupa tanggal polos // (mis. kegiatan default sebelum pernah diklik "+Kegiatan/Sesi Baru"). // Sekarang pencocokan HANYA memakai isi kolom IDKegiatan APA ADANYA (exact // match, tanpa fallback), supaya hapus kegiatan tidak pernah menyapu baris // lain yang kebetulan berbagi ID fallback yang sama. function deleteEvent_(ss, body) { var eventId = body.eventId; if (!eventId) return; var settingsSheet = ss.getSheetByName('Pengaturan'); if (settingsSheet) { var sData = settingsSheet.getDataRange().getValues(); for (var i = sData.length - 1; i >= 1; i--) { if (String(sData[i][PENGATURAN_COL.ID_KEGIATAN]) === String(eventId)) { settingsSheet.deleteRow(i + 1); } } } var pesertaSheet = ss.getSheetByName('Peserta'); if (pesertaSheet) { var pData = pesertaSheet.getDataRange().getValues(); for (var j = pData.length - 1; j >= 1; j--) { if (String(pData[j][COL.ID_KEGIATAN]) === String(eventId)) { pesertaSheet.deleteRow(j + 1); } } } } // ================== DAFTAR PESERTA (tab "Peserta") ================== var PESERTA_HEADERS = [ 'ID', 'Tanggal', 'Kegiatan', 'Lokasi', 'Nama', 'Gender', 'Institusi', 'Jabatan', 'Telepon', 'BukanDisabilitas', 'Disabilitas', 'DisabilitasLainnya', 'TolakSebutkanDisabilitas', 'TandaTanganURL', 'WaktuAbsen', 'IDKegiatan' ]; var COL = { ID:0, TANGGAL:1, KEGIATAN:2, LOKASI:3, NAMA:4, GENDER:5, INSTITUSI:6, JABATAN:7, TELEPON:8, BUKAN_DISABILITAS:9, DISABILITAS:10, DISABILITAS_LAINNYA:11, TOLAK_SEBUTKAN:12, TTD_URL:13, WAKTU_ABSEN:14, ID_KEGIATAN:15 }; function getParticipantsForEvent_(ss, eventId) { var sheet = ss.getSheetByName('Peserta'); if (!sheet) return []; var data = sheet.getDataRange().getValues(); var list = []; for (var i = 1; i < data.length; i++) { if (rowEventId_(data[i], COL.TANGGAL, COL.ID_KEGIATAN) === eventId) { list.push({ id: data[i][COL.ID], name: data[i][COL.NAMA], gender: data[i][COL.GENDER], institution: data[i][COL.INSTITUSI], role: data[i][COL.JABATAN], phone: data[i][COL.TELEPON], notDisabled: data[i][COL.BUKAN_DISABILITAS] === true || data[i][COL.BUKAN_DISABILITAS] === 'TRUE', disabilities: data[i][COL.DISABILITAS] ? String(data[i][COL.DISABILITAS]).split(',').map(function(s){ return s.trim(); }).filter(Boolean) : [], disabilityOther: data[i][COL.DISABILITAS_LAINNYA] || '', declineDisabilityInfo: data[i][COL.TOLAK_SEBUTKAN] === true || data[i][COL.TOLAK_SEBUTKAN] === 'TRUE', signature: data[i][COL.TTD_URL] || null }); } } return list; } function saveParticipant_(ss, body) { var sheet = ss.getSheetByName('Peserta'); if (!sheet) { sheet = ss.insertSheet('Peserta'); sheet.appendRow(PESERTA_HEADERS); } var eventId = body.eventId || body.tanggal; var data = sheet.getDataRange().getValues(); var rowIndex = -1; for (var i = 1; i < data.length; i++) { if (String(data[i][COL.ID]) === String(body.id)) { rowIndex = i + 1; break; } } var signatureUrl = ''; if (body.clearSignature) { signatureUrl = ''; } else if (body.signature) { signatureUrl = saveSignatureToDrive_(body.id, body.tanggal, body.signature); } else if (rowIndex > 0) { signatureUrl = data[rowIndex - 1][COL.TTD_URL] || ''; } var row = [ body.id, body.tanggal, body.kegiatan, body.lokasi, body.nama, body.gender, body.institusi, body.jabatan, body.telepon, body.notDisabled, body.disabilitas, body.disabilitasLainnya, body.tolakSebutkan, signatureUrl, body.waktuAbsen || (rowIndex > 0 ? data[rowIndex - 1][COL.WAKTU_ABSEN] : ''), eventId ]; if (rowIndex > 0) { sheet.getRange(rowIndex, 1, 1, row.length).setValues([row]); } else { sheet.appendRow(row); } } function deleteParticipant_(ss, body) { var sheet = ss.getSheetByName('Peserta'); if (!sheet) return; var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][COL.ID]) === String(body.id)) { sheet.deleteRow(i + 1); break; } } } // Pindahkan peserta ke KEGIATAN lain (bisa beda tanggal, atau kegiatan lain di // tanggal yang sama). Mengubah kolom Tanggal, IDKegiatan, Kegiatan & Lokasi; // field lain (nama, institusi, TTD, dst) tetap dipertahankan apa adanya. function moveParticipantDate_(ss, body) { var sheet = ss.getSheetByName('Peserta'); if (!sheet) return; var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][COL.ID]) === String(body.id)) { var rowNum = i + 1; sheet.getRange(rowNum, COL.TANGGAL + 1).setValue(body.newDate); if (body.newEventId) sheet.getRange(rowNum, COL.ID_KEGIATAN + 1).setValue(body.newEventId); if (body.newKegiatan !== undefined) sheet.getRange(rowNum, COL.KEGIATAN + 1).setValue(body.newKegiatan); if (body.newLokasi !== undefined) sheet.getRange(rowNum, COL.LOKASI + 1).setValue(body.newLokasi); break; } } } // Upload TTD (base64 PNG) ke Google Drive, kembalikan URL yang reliable dijadikan . function saveSignatureToDrive_(participantId, date, base64DataUrl) { if (!base64DataUrl) return ''; var matches = base64DataUrl.match(/^data:(image\/\w+);base64,(.*)$/); if (!matches) return ''; var mimeType = matches[1]; var base64 = matches[2]; var bytes = Utilities.base64Decode(base64); var blob = Utilities.newBlob(bytes, mimeType, 'ttd_' + date + '_' + participantId + '.png'); var folder = getOrCreateFolder_(SIGNATURE_FOLDER_NAME); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return toDirectImageUrl_(file.getId()); } // Format lh3.googleusercontent.com jauh lebih konsisten untuk dijadikan // langsung dibanding link "uc?export=view" atau "thumbnail" dari Drive. function toDirectImageUrl_(fileId) { return 'https://lh3.googleusercontent.com/d/' + fileId + '=w1000'; } function getOrCreateFolder_(name) { var folders = DriveApp.getFoldersByName(name); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(name); } // Ambil fileId Drive dari URL TTD lama, format apa pun (v3 "uc?export=view&id=", // v4 "thumbnail?id=", atau sudah v5 "lh3.googleusercontent.com/d/ID=..."). function extractDriveFileId_(url) { if (typeof url !== 'string' || !url) return null; var m = url.match(/[?&]id=([a-zA-Z0-9_-]+)/); if (m) return m[1]; m = url.match(/lh3\.googleusercontent\.com\/d\/([a-zA-Z0-9_-]+)/); if (m) return m[1]; return null; } // ================== MIGRASI LINK TTD LAMA (jalankan manual SEKALI SAJA) ================== function migrateSignatureLinksToLh3() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Peserta'); if (!sheet) { Logger.log('Tab "Peserta" belum ada, tidak ada yang perlu dimigrasi.'); return; } var data = sheet.getDataRange().getValues(); var updated = 0; for (var i = 1; i < data.length; i++) { var url = data[i][COL.TTD_URL]; if (typeof url === 'string' && url && url.indexOf('lh3.googleusercontent.com') === -1) { var fileId = extractDriveFileId_(url); if (fileId) { sheet.getRange(i + 1, COL.TTD_URL + 1).setValue(toDirectImageUrl_(fileId)); updated++; } } } Logger.log('Selesai. ' + updated + ' link TTD berhasil dimigrasi ke format lh3.googleusercontent.com.'); } // ================== RIWAYAT KEHADIRAN (format lama, tetap dipakai sebagai log) ================== function saveAttendance_(ss, body) { var sheet = ss.getSheetByName('Kehadiran') || ss.getSheets()[0]; sheet.appendRow([ body.kegiatan, body.tanggal, body.lokasi, body.nama, body.gender, body.institusi, body.jabatan, body.telepon, body.disabilitas, body.waktuAbsen ]); } // ================== HELPER ================== function formatDate_(value) { if (!value) return ''; var d = (value instanceof Date) ? value : new Date(value); if (isNaN(d.getTime())) return String(value); return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } function jsonResponse_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
